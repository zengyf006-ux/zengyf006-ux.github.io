ba(245,180,95,.11)';ctx.beginPath();ctx.arc(W*.5,H-105,52,Math.PI,TAU);ctx.stroke();}
function updateEffects(dt){for(let i=particles.length-1;i>=0;i--){const p=particles[i];p.x+=p.vx;p.y+=p.vy;p.vx*=.984;p.vy=p.vy*.984+.035;p.life-=dt*1.7;if(p.life<=0){particles.splice(i,1);continue;}ctx.globalAlpha=p.life;ctx.fillStyle=p.color;ctx.beginPath();ctx.arc(p.x,p.y,p.size*p.life,0,TAU);ctx.fill();}ctx.globalAlpha=1;for(let i=rings.length-1;i>=0;i--){const r=rings[i];r.life-=dt*1.6;r.r+=dt*82;if(r.life<=0){rings.splice(i,1);continue;}ctx.globalAlpha=r.life;ctx.strokeStyle=r.color;ctx.lineWidth=r.w;ctx.beginPath();ctx.arc(r.x,r.y,r.r,0,TAU);ctx.stroke();}ctx.globalAlpha=1;for(let i=trails.length-1;i>=0;i--){const t=trails[i];t.life-=dt*(t.lightning?4.2:3.8);if(t.life<=0){trails.splice(i,1);continue;}ctx.globalAlpha=t.life;ctx.strokeStyle=t.lightning?colors.cyan:colors.amber;ctx.lineWidth=t.lightning?2:2+6*t.life;ctx.lineCap='round';ctx.beginPath();ctx.moveTo(t.ax,t.ay);if(t.lightning){const mx=(t.ax+t.bx)/2+rand(-8,8),my=(t.ay+t.by)/2+rand(-8,8);ctx.lineTo(mx,my);ctx.lineTo(t.bx,t.by);}else ctx.lineTo(t.bx,t.by);ctx.stroke();}ctx.globalAlpha=1;for(let i=texts.length-1;i>=0;i--){const t=texts[i];t.y+=t.vy*dt;t.life-=dt*1.35;if(t.life<=0){texts.splice(i,1);continue;}ctx.globalAlpha=t.life;ctx.fillStyle=t.color;ctx.font='900 '+t.size+'px -apple-system,BlinkMacSystemFont,"PingFang SC"';ctx.textAlign='center';ctx.textBaseline='middle';ctx.fillText(t.text,t.x,t.y);}ctx.globalAlpha=1;}
function updateGame(dt){
 if(state.paused)return;
 state.waveTime+=dt;if(speechCooldown>0)speechCooldown-=dt;
 if(state.comboTimer>0){state.comboTimer-=dt;if(state.comboTimer<=0){state.chain=0;state.combo=1;}}
 if(state.skillCooldown>0)state.skillCooldown-=dt;
 if(state.feverUntil>0&&clock>=state.feverUntil){state.feverUntil=0;state.fever=0;document.getElementById('app').classList.remove('fever');message('狂热结束');}
 for(const m of monsters)if(!m.dead)m.update(dt);
 for(let i=monsters.length-1;i>=0;i--)if(monsters[i].dead)monsters.splice(i,1);
 if(boss)updateBoss(dt);
 const idx=(state.wave-1)%4;
 if(idx!==3&&!state.transitionPending){
  const alive=monsters.length;
  const desired=Math.min(20,Math.round((6+Math.floor(state.wave*.7))*currentModifier.spawn));
  if(!state.waveEnding){
   state.spawnClock-=dt;
   if(state.spawnClock<=0&&alive<desired){spawnMonster(null,state.wave>2&&Math.random()<.08);state.spawnClock=Math.max(.2,(.72-state.wave*.018)/currentModifier.spawn);}
   if(state.waveTime>=state.waveDuration){state.waveEnding=true;ui.mission.textContent='最后清理：击杀场内残余怪物';message('最后清理',900);}
  }else if(alive===0||state.waveTime>=state.waveDuration+7){endWave();}
 }
}
