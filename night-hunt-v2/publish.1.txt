function endWaveOnce(state) {
  if (state.waveCleared || state.transitionPending) return false;
  state.waveCleared = true;
  state.transitionPending = true;
  state.wave += 1;
  return true;
}

function applyBossDirectDamage(boss, damage) {
  if (!boss || boss.dead) return { killed: false, damage: 0 };
  const dealt = Math.max(0, Number.isFinite(damage) ? damage : 0);
  boss.hp = Math.max(0, boss.hp - dealt);
  if (boss.hp === 0) boss.dead = true;
  return { killed: boss.dead, damage: dealt };
}

function pickUpgradeOptions(upgrades, levels, fallbacks, count = 3, random = Math.random) {
  const available = upgrades.filter(u => (levels[u.id] || 0) < u.max);
  const pool = [...available];
  const picks = [];
  while (pool.length && picks.length < count) {
    const index = Math.min(pool.length - 1, Math.floor(random() * pool.length));
    picks.push(pool.splice(index, 1)[0]);
  }
  for (const fallback of fallbacks) {
    if (picks.length >= count) break;
    picks.push(fallback);
  }
  return picks.slice(0, count);
}

function childSpawnCount(current, cap, requested) {
  return Math.max(0, Math.min(requested, cap - current));
}

function applyEscapePenalty(type, skill, gate) {
  const damage = 1;
  return {
    skill: Math.max(0, skill - (type === 'leech' ? 12 : 0)),
    gate: Math.max(0, gate - damage),
    damage
  };
}

function resolveEvolution(levels) {
  if ((levels.damage || 0) >= 3 && (levels.width || 0) >= 2) return '月蚀巨刃';
  if ((levels.chain || 0) >= 2 && (levels.crit || 0) >= 2) return '雷狱猎刃';
  if ((levels.pulse || 0) >= 2 && (levels.skill || 0) >= 2) return '虚空震刃';
  return null;
}

function sanitizeSnapshot(raw = {}) {
  const finiteInt = (value, fallback, min = 0) => {
    const n = Number(value);
    return Number.isFinite(n) ? Math.max(min, Math.floor(n)) : fallback;
  };
  return {
    bestScore: finiteInt(raw.bestScore, 0, 0),
    bestWave: finiteInt(raw.bestWave, 1, 1),
    totalKills: finiteInt(raw.totalKills, 0, 0),
    sound: typeof raw.sound === 'boolean' ? raw.sound : true,
    voice: typeof raw.voice === 'boolean' ? raw.voice : true
  };
}
